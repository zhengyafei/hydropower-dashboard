function selectStation(stationId) {
  _treeSelectedStation = stationId;
  buildOrgCascaderList();
  showStationView(stationId);
}

function drawStationCharts(){
  var sd = window._STATION_DATA || {};
  var devDetail = (sd.deviceDetail||{}).body||[];
  var weatherNearest = (sd.weatherNearest||{}).body||[];

  // Mock 30-day data if API data is empty
  function mock30Day() {
    var dates = []; var vals = [];
    var now = new Date();
    for (var i = 29; i >= 0; i--) {
      var d = new Date(now); d.setDate(d.getDate()-i);
      dates.push((d.getMonth()+1)+'/'+d.getDate());
      vals.push(Math.round(Math.random()*8000+2000));
    }
    return {dates:dates, vals:vals};
  }

  // Mock hourly data for real-time chart
  function mockHourly() {
    var hours = []; var vals = [];
    for (var i = 0; i < 24; i++) {
      hours.push(i+':00');
      vals.push(Math.round(Math.random()*400+100));
    }
    return {hours:hours, vals:vals};
  }

  // Mock monthly data
  function mockMonthly() {
    var months = ['1月','2月','3月','4月','5月','6月','7月','8月','9月','10月','11月','12月'];
    var fdl = months.map(function(){return (Math.random()*60+10).toFixed(2);});
    var rain = months.map(function(){return (Math.random()*200+20).toFixed(1);});
    return {months:months, fdl:fdl, rain:rain};
  }

  // --- Chart 1: 实时发电量/降雨量趋势 ---
  var elRT = document.getElementById('chartStRealtime');
  if (elRT) {
    if (window._stRT) window._stRT.dispose();
    var mh = mockHourly();
    var chart = echarts.init(elRT); window._stRT = chart;
    chart.setOption({
      tooltip: {trigger:'axis'},
      legend: {data:['实时发电量','降雨量'],textStyle:{fontSize:10,color:'#8899b4'},top:0},
      grid: {left:50,right:50,top:30,bottom:24},
      xAxis: {type:'category',data:mh.hours,axisLabel:{fontSize:9,color:'#5a6d84'},axisLine:{lineStyle:{color:'rgba(255,255,255,0.08)'}}},
      yAxis: [
        {type:'value',name:'kW',axisLabel:{fontSize:9,color:'#5a6d84'},splitLine:{lineStyle:{color:'rgba(255,255,255,0.04)'}}},
        {type:'value',name:'mm',axisLabel:{fontSize:9,color:'#5a6d84'},splitLine:{show:false}}
      ],
      series: [
        {name:'实时发电量',type:'line',data:mh.vals,lineStyle:{color:'#1e90ff',width:2},itemStyle:{color:'#1e90ff'},symbol:'circle',symbolSize:3,areaStyle:{color:'rgba(30,144,255,0.1)'}},
        {name:'降雨量',type:'bar',yAxisIndex:1,data:mh.vals.map(function(){return (Math.random()*1.5).toFixed(1);}),itemStyle:{color:'#00b4d8'},barMaxWidth:8,barGap:'20%'}
      ]
    });
  }

  // --- Chart 2: 近期发电量/降雨量趋势 (30-day) ---
  var el30 = document.getElementById('chartStRecent');
  if (el30) {
    if (window._stRecent) window._stRecent.dispose();
    var fdlS = null, pcpnS = null;
    (weatherNearest||[]).forEach(function(s){if(s.type==='fdl')fdlS=s;if(s.type==='pcpn')pcpnS=s;});
    var fx, fy, py;
    if (fdlS && fdlS.x && fdlS.x.length > 0) {
      fx = fdlS.x.map(function(d){return d.substring(5);});
      fy = (fdlS.y||[]).map(function(v){return parseFloat(v)||0;});
      py = (pcpnS&&pcpnS.y||[]).map(function(v){return parseFloat(v)||0;});
    } else {
      var m30 = mock30Day();
      fx = m30.dates; fy = m30.vals; py = m30.vals.map(function(){return (Math.random()*8).toFixed(1);});
    }
    var chart = echarts.init(el30); window._stRecent = chart;
    chart.setOption({
      tooltip: {trigger:'axis'},
      legend: {data:['发电量','降雨量'],textStyle:{fontSize:10,color:'#8899b4'},top:0},
      grid: {left:50,right:50,top:30,bottom:24},
      xAxis: {type:'category',data:fx,axisLabel:{fontSize:9,color:'#5a6d84',rotate:30},axisLine:{lineStyle:{color:'rgba(255,255,255,0.08)'}}},
      yAxis: [
        {type:'value',name:'kWh',axisLabel:{fontSize:9,color:'#5a6d84'},splitLine:{lineStyle:{color:'rgba(255,255,255,0.04)'}}},
        {type:'value',name:'mm',axisLabel:{fontSize:9,color:'#5a6d84'},splitLine:{show:false}}
      ],
      series: [
        {name:'发电量',type:'bar',data:fy,itemStyle:{color:'#1e90ff'},barMaxWidth:12},
        {name:'降雨量',type:'line',yAxisIndex:1,data:py,lineStyle:{color:'#00b4d8',width:2},itemStyle:{color:'#00b4d8'},symbol:'circle',symbolSize:4}
      ]
    });
  }

  // --- Chart 3: 月度发电量/降雨量对比 ---
  var elMon = document.getElementById('chartStMonthly');
  if (elMon) {
    if (window._stMonthly) window._stMonthly.dispose();
    var mm = mockMonthly();
    var chart = echarts.init(elMon); window._stMonthly = chart;
    chart.setOption({
      tooltip: {trigger:'axis'},
      legend: {data:['月发电量(万kWh)','月降雨量(mm)'],textStyle:{fontSize:10,color:'#8899b4'},top:0},
      grid: {left:50,right:50,top:30,bottom:24},
      xAxis: {type:'category',data:mm.months,axisLabel:{fontSize:9,color:'#5a6d84'},axisLine:{lineStyle:{color:'rgba(255,255,255,0.08)'}}},
      yAxis: [
        {type:'value',name:'万kWh',axisLabel:{fontSize:9,color:'#5a6d84'},splitLine:{lineStyle:{color:'rgba(255,255,255,0.04)'}}},
        {type:'value',name:'mm',axisLabel:{fontSize:9,color:'#5a6d84'},splitLine:{show:false}}
      ],
      series: [
        {name:'月发电量(万kWh)',type:'bar',data:mm.fdl,itemStyle:{color:'#1e90ff'},barMaxWidth:20,barGap:'20%'},
        {name:'月降雨量(mm)',type:'bar',data:mm.rain,itemStyle:{color:'#48cae4'},barMaxWidth:20}
      ]
    });
  }

  // --- Unit 30-day charts ---
  var unitCount = (devDetail||[]).length || 2;
  for (var ui = 0; ui < unitCount; ui++) {
    var el = document.getElementById('stUnitChart'+ui);
    if (!el) continue;
    var fdl = (devDetail[ui]||{}).fdl || {};
    var dates = fdl.x || [];
    var vals = (fdl.y || []).map(function(v){return parseFloat(v)||0;});

    // If no API data, use mock
    if (dates.length === 0) {
      var m30 = mock30Day();
      dates = m30.dates.map(function(d){return '2026-'+d.replace('/','-');});
      vals = m30.vals;
    }

    var lastIdx = vals.length - 1;
    while (lastIdx >= 0 && vals[lastIdx] === 0) lastIdx--;
    if (lastIdx < 0) lastIdx = vals.length - 1;
    var start = Math.max(0, lastIdx - 29);
    var xd = dates.slice(start, lastIdx + 1).map(function(d){return d.substring(5);});
    var yd = vals.slice(start, lastIdx + 1);

    if (window['_stUC'+ui]) window['_stUC'+ui].dispose();
    var chart = echarts.init(el); window['_stUC'+ui] = chart;
    chart.setOption({
      grid: {left:40,right:12,top:8,bottom:24},
      xAxis: {type:'category',data:xd,axisLabel:{fontSize:9,color:'#5a6d84',rotate:30},axisLine:{lineStyle:{color:'rgba(255,255,255,0.08)'}}},
      yAxis: {type:'value',axisLabel:{fontSize:9,color:'#5a6d84'},splitLine:{lineStyle:{color:'rgba(255,255,255,0.04)'}}},
      series: [{type:'line',data:yd,lineStyle:{color:'#1e90ff',width:2},itemStyle:{color:'#1e90ff'},symbol:'circle',symbolSize:3,areaStyle:{color:'rgba(30,144,255,0.08)'}}],
      tooltip: {trigger:'axis',textStyle:{fontSize:11}}
    });
  }

  // --- Unit run record charts (mock) ---
  for (var ri = 0; ri < unitCount; ri++) {
    var rel = document.getElementById('stRunChart'+ri);
    if (!rel) continue;
    if (window['_stRun'+ri]) window['_stRun'+ri].dispose();
    var categories = ['开机','发电','停机','待用'];
    var mockDurations = ri===0 ? [680,13640,220,1380] : [620,520,40,380];
    mockDurations = mockDurations.map(function(v){return v/3600;}); // Convert seconds to hours
    var chart = echarts.init(rel); window['_stRun'+ri] = chart;
    chart.setOption({
      grid: {left:50,right:10,top:4,bottom:20},
      yAxis: {type:'category',data:categories,axisLabel:{fontSize:10,color:'#8899b4'},axisLine:{lineStyle:{color:'rgba(255,255,255,0.08)'}}},
      xAxis: {type:'value',name:'h',axisLabel:{fontSize:9,color:'#5a6d84'},splitLine:{lineStyle:{color:'rgba(255,255,255,0.04)'}}},
      series: [{
        type:'bar',data:mockDurations,
        itemStyle:{color:function(p){return ['#ff9100','#1e90ff','#546e7a','#4caf50'][p.dataIndex];}},
        barMaxWidth:16,
        label:{show:true,position:'right',formatter:function(p){return p.value.toFixed(1)+'h';},fontSize:10,color:'#8899b4'}
      }],
      tooltip: {trigger:'axis'}
    });
  }
}
